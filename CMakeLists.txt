cmake_minimum_required(VERSION 3.20)
project(batched_dag_cuda VERSION 0.1.0 LANGUAGES CXX)

option(BDG_ENABLE_CUDA "Build CUDA backends when a CUDA compiler is available" ON)
option(BDG_REQUIRE_CUDA "Fail configuration if the CUDA backend cannot be built" OFF)
option(BDG_ENABLE_SANITIZERS "Use ASan and UBSan for a CPU-only development build" OFF)
option(BDG_CUDA_PTXAS_INFO "Print CUDA register and spill information" OFF)
include(CTest)
include(GNUInstallDirs)

set(BDG_HAS_CUDA OFF)
if(BDG_REQUIRE_CUDA AND NOT BDG_ENABLE_CUDA)
    message(FATAL_ERROR "BDG_REQUIRE_CUDA conflicts with BDG_ENABLE_CUDA=OFF")
endif()
if(BDG_ENABLE_CUDA)
    include(CheckLanguage)
    check_language(CUDA)
    if(CMAKE_CUDA_COMPILER)
        if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
            set(CMAKE_CUDA_ARCHITECTURES 80 CACHE STRING "CUDA architectures")
        endif()
        enable_language(CUDA)
        find_package(CUDAToolkit 12.0 REQUIRED)
        set(BDG_HAS_CUDA ON)
    elseif(BDG_REQUIRE_CUDA)
        message(FATAL_ERROR "A CUDA compiler is required but was not found")
    else()
        message(STATUS "No CUDA compiler: building CPU reference and explicit CUDA-unavailable stub")
    endif()
endif()
if(BDG_ENABLE_SANITIZERS AND BDG_HAS_CUDA)
    message(FATAL_ERROR "Use a separate CPU-only build for BDG_ENABLE_SANITIZERS")
endif()

add_library(bdg_core src/graph.cpp src/io.cpp src/plan.cpp)
add_library(bdg::core ALIAS bdg_core)
set_target_properties(bdg_core PROPERTIES EXPORT_NAME core)
target_compile_features(bdg_core PUBLIC cxx_std_17)
target_include_directories(bdg_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
                                         $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(MSVC)
    target_compile_options(bdg_core PRIVATE /W4)
else()
    target_compile_options(bdg_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BDG_ENABLE_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "Sanitizer option currently supports GCC and Clang")
    endif()
    target_compile_options(bdg_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bdg_core PUBLIC -fsanitize=address,undefined)
endif()

if(BDG_HAS_CUDA)
    add_library(bdg_cuda src/cuda_common.cpp src/cuda_executor.cu)
    target_link_libraries(bdg_cuda PUBLIC bdg_core PRIVATE CUDA::cudart)
    set_target_properties(bdg_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES
                                             CUDA_SEPARABLE_COMPILATION OFF)
    target_compile_options(bdg_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
    if(BDG_CUDA_PTXAS_INFO)
        target_compile_options(bdg_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xptxas=-v,-warn-spills>)
    endif()
else()
    add_library(bdg_cuda src/cuda_common.cpp src/cuda_stub.cpp)
    target_link_libraries(bdg_cuda PUBLIC bdg_core)
endif()
add_library(bdg::cuda ALIAS bdg_cuda)
set_target_properties(bdg_cuda PROPERTIES EXPORT_NAME cuda)
target_compile_definitions(bdg_cuda PUBLIC BDG_HAS_CUDA=$<BOOL:${BDG_HAS_CUDA}>)

add_library(bdg_sample STATIC examples/cos_sample.cpp)
target_link_libraries(bdg_sample PUBLIC bdg_core)
target_include_directories(bdg_sample PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/examples)
add_executable(bdg_make_cos_fixture apps/make_cos_fixture.cpp)
target_link_libraries(bdg_make_cos_fixture PRIVATE bdg_sample)
add_executable(bdg_benchmark apps/benchmark.cpp)
target_link_libraries(bdg_benchmark PRIVATE bdg_cuda)
target_compile_definitions(bdg_benchmark PRIVATE BDG_VERSION="${PROJECT_VERSION}")
add_executable(bdg_evaluate examples/evaluate.cpp)
target_link_libraries(bdg_evaluate PRIVATE bdg_cuda)

if(BUILD_TESTING)
    add_executable(bdg_tests tests/cpu_tests.cpp)
    target_link_libraries(bdg_tests PRIVATE bdg_sample)
    target_compile_definitions(bdg_tests PRIVATE BDG_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
    add_test(NAME cpu_correctness COMMAND bdg_tests)
    add_executable(bdg_cuda_tests tests/cuda_tests.cpp)
    target_link_libraries(bdg_cuda_tests PRIVATE bdg_cuda bdg_sample)
    add_test(NAME cuda_correctness COMMAND bdg_cuda_tests)
    set_tests_properties(cuda_correctness PROPERTIES SKIP_RETURN_CODE 77 LABELS gpu TIMEOUT 180)
    add_test(NAME cli_help COMMAND bdg_benchmark --help)
    add_test(NAME cli_cpu COMMAND bdg_benchmark --graph ${CMAKE_CURRENT_SOURCE_DIR}/data/diamond.dag
             --backend cpu --batch 13 --iterations 3 --warmup 1)
    add_test(NAME cli_batch_file COMMAND bdg_benchmark --graph ${CMAKE_CURRENT_SOURCE_DIR}/data/diamond.dag
             --input ${CMAKE_CURRENT_SOURCE_DIR}/data/diamond.batch --backend cpu --iterations 2 --warmup 0)
    add_test(NAME cli_reject_zero_batch COMMAND bdg_benchmark --graph ${CMAKE_CURRENT_SOURCE_DIR}/data/diamond.dag --batch 0)
    set_tests_properties(cli_reject_zero_batch PROPERTIES WILL_FAIL TRUE)
    add_test(NAME cli_cos_fixture COMMAND bdg_benchmark --graph ${CMAKE_CURRENT_SOURCE_DIR}/data/cos_su2_n2.dag
             --backend cpu --batch 13 --iterations 2 --warmup 0)
endif()

install(TARGETS bdg_core bdg_cuda EXPORT bdgTargets
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
        FILES_MATCHING PATTERN "*.hpp" PATTERN "._*" EXCLUDE)
install(EXPORT bdgTargets NAMESPACE bdg:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/bdg)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/bdgConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/bdgConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/bdg)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/bdgConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/bdgConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/bdgConfigVersion.cmake
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/bdg)
message(STATUS "batched-dag-cuda: CUDA backend=${BDG_HAS_CUDA}")
